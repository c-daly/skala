cmake_minimum_required(VERSION 3.16)
project(skala_rviz_teleop)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_library(skala_drive_panel SHARED src/drive_panel.cpp)
target_compile_features(skala_drive_panel PUBLIC cxx_std_17)
target_link_libraries(skala_drive_panel Qt5::Widgets)
ament_target_dependencies(skala_drive_panel rviz_common pluginlib rclcpp geometry_msgs std_msgs)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS skala_drive_panel LIBRARY DESTINATION lib)
ament_package()
